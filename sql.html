-- ===== Respostas dos formulários da Nexum (empresa de programação Elton Longaray) =====
create table if not exists public.nexum_respostas (
  id bigint generated always as identity primary key,
  setor text not null check (setor in ('delce','recepcao','medicos','contador')),
  nome text not null check (char_length(nome) between 1 and 120),
  respostas jsonb not null check (pg_column_size(respostas) < 40000),
  criado_em timestamptz not null default now()
);

alter table public.nexum_respostas enable row level security;

-- Qualquer pessoa com o link pode ENVIAR uma resposta (não consegue ler nem apagar)
drop policy if exists "nexum envia" on public.nexum_respostas;
create policy "nexum envia" on public.nexum_respostas
  for insert to anon, authenticated with check (true);
